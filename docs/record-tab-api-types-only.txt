// 기록 탭 최소 API 타입 정리
// 역할별 제목만 붙인 붙여넣기용 메모

// 1. 라이브 세션 시작 요청
type LiveSessionStartRequest = {
  uuid: string;
  dance_type: string;
  content_id: string;
};

// 2. 라이브 세션 시작 응답
type LiveSessionStartResponse = {
  session_id: string;
  ws_url?: string;
};

// 3. 실시간 프레임 전송 메타데이터
type LiveFrameBinaryMeta = {
  type: 'frame_binary';
  UUID: string;
  session_id: string;
  frame_index: number;
  total_frame: number;
  user_weight: number;
};

// 4. 실시간 분석 결과 메시지
type LiveFrameResultMessage = {
  type: 'frame_result' | 'frame_ack';
  session_id: string;
  total_frames: number;
  elapsed_seconds: number;
  total_calories: number;
  movement_score: number;
};

// 5. 라이브 세션 종료 요청
type LiveSessionEndRequest = {
  session_id: string;
};

// 6. 라이브 세션 종료 응답
type LiveSessionEndResponse = {
  ok: boolean;
  session_id: string;
};

// 7. 운동 세션 DB 저장 행
type ExerciseSessionRow = {
  session_id: string;
  user_id: string;
  class_id: string | null;
  dance_name?: string | null;
  genre: string | null;
  started_at: string;
  ended_at: string;
  duration_seconds: number;
  burned_kcal: number;
  movement_score?: number | null;
  total_frames?: number | null;
  user_weight_kg?: number | null;
  created_at?: string;
};

// 8. 기록 탭 월간 조회 쿼리
type RecordsQuery = {
  userId: string;
  year: number;
  month: number;
};

// 9. 기록 탭 월간 요약
type RecordSummary = {
  totalKcal: number;
  totalMinutes: number;
  activeDays: number;
  sessionCount: number;
  avgKcalPerSession: number;
  peakWeekLabel: string;
  peakWeekKcal: number;
};

// 10. 기록 탭 캘린더 일별 집계
type RecordCalendarItem = {
  dateKey: string;
  day: number;
  kcal: number;
  minutes: number;
  count: number;
};

// 11. 기록 탭 세션 목록 아이템
type RecordSessionItem = {
  sessionId: string;
  dateKey: string;
  performedAt: string;
  danceName: string;
  danceType: string | null;
  kcal: number;
  durationMinutes: number;
  movementScore: number | null;
};

// 12. 기록 탭 월간 조회 응답
type RecordsResponse = {
  summary: RecordSummary;
  calendar: RecordCalendarItem[];
  sessions: RecordSessionItem[];
};

// 13. 체중 추이 조회 쿼리
type WeightHistoryQuery = {
  userId: string;
  months: number;
};

// 14. 체중 추이 히스토리 아이템
type WeightHistoryItem = {
  dateKey: string;
  year: number;
  month: number;
  weightKg: number;
  note: string | null;
};

// 15. 체중 추이 조회 응답
type WeightHistoryResponse = {
  currentWeightKg: number | null;
  targetWeightKg: number | null;
  history: WeightHistoryItem[];
};

// 16. 프로필 fallback 응답
type ProfileFallbackResponse = {
  uuid: string;
  name: string;
  email: string;
  age: number;
  password: string;
  created_at: string;
  height: number | null;
  weight: number | null;
  target_weight: number | null;
  target_day: string | null;
  target_kcal: number | null;
  today_target_kcal: number | null;
  bucket_profile_photo: string | null;
  filepath: string | null;
  profile_image_url: string | null;
  current_streak: number | null;
};
